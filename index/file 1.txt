const DISTRICTS = ['All', 'Brunei-Muara', 'Tutong', 'Belait', 'Temburong'];

export default function App() {
  const [showSplash, setShowSplash] = useState(true);
  const [search, setSearch] = useState('');
  const [district, setDistrict] = useState('All');
  const [selected, setSelected] = useState(null);
  const [favs, setFavs] = useState([]);
  const [tab, setTab] = useState('home');

  useEffect(() => {
    const t = setTimeout(() => setShowSplash(false), 2200);
    return () => clearTimeout(t);
  }, []);

  if (showSplash) {
    return (
      <View style={{ flex: 1, backgroundColor: '#FAF7F2', alignItems: 'center', justifyContent: 'center' }}>
        <Image source={require('./assets/logo.png')} style={{ width: 230, height: 230, resizeMode: 'contain' }} />
        <Text style={{ marginTop: 8, fontSize: 10, letterSpacing: 3, color: '#A08B7A' }}>BRUNEI • EST. 2026</Text>
        <Text style={{ marginTop: 6, fontSize: 13, color: '#8A7A6D' }}>Small cafes, quiet places.</Text>
      </View>
    );
  }

  const filtered = CAFES.filter(c => {
    const s = (c.name + c.area).toLowerCase().includes(search.toLowerCase());
    const d = district === 'All' || c.district === district;
    const f = tab === 'home' || favs.includes(c.id);
    return s && d && f;
  });

  const toggleFav = (id) => setFavs(favs.includes(id) ? favs.filter(f => f !== id) : [...favs, id]);

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="dark-content" />
      <View style={styles.top}>
        <Text style={styles.smallLabel}>BRUNEI • EST. 2026</Text>
        <Text style={styles.title}>Café{"\n"}Finders.</Text>
        <Text style={styles.subtitle}>Small cafes, quiet places, and hidden gems across Brunei.</Text>
        <View style={styles.searchWrap}>
          <Text style={{ fontSize: 16, color: '#A08B7A' }}>⌕</Text>
          <TextInput style={styles.search} placeholder="Search a cafe, area, matcha..." placeholderTextColor="#B4A79B" value={search} onChangeText={setSearch} />
        </View>
      </View>
      <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.filterRow} contentContainerStyle={{ paddingHorizontal: 24 }}>
        {DISTRICTS.map(d => (
          <TouchableOpacity key={d} onPress={() => setDistrict(d)}>
            <Text style={[styles.filterText, district === d && styles.filterActive]}>{d}</Text>
            {district === d && <View style={styles.filterDot} />}
          </TouchableOpacity>
        ))}
      </ScrollView>
      <Text style={styles.count}>{filtered.length} places — {tab === 'fav' ? 'Your favourites' : 'Curated for you'}</Text>
      <FlatList data={filtered} keyExtractor={i => i.id} showsVerticalScrollIndicator={false} contentContainerStyle={{ paddingHorizontal: 24, paddingBottom: 110 }}
        renderItem={({ item }) => (
          <TouchableOpacity style={styles.card} onPress={() => setSelected(item)} activeOpacity={0.9}>
            <Image source={{ uri: item.img }} style={styles.cardImg} />
            <View style={styles.cardTop}>
              <View style={styles.tag}><Text style={styles.tagText}>{item.tag}</Text></View>
              <TouchableOpacity onPress={() => toggleFav(item.id)} style={styles.heart}><Text style={{ fontSize: 16 }}>{favs.includes(item.id) ? '♥' : '♡'}</Text></TouchableOpacity>
            </View>
            <View style={styles.cardBody}>
              <Text style={styles.cardName}>{item.name}</Text>
              <Text style={styles.cardArea}>{item.area} • {item.open}</Text>
              <View style={styles.divider} />
              <View style={{ flexDirection: 'row', justifyContent: 'space-between' }}>
                <Text style={styles.cardMeta}>★ {item.rating} • {item.price} • {item.cat}</Text>
                <Text style={styles.viewText}>View →</Text>
              </View>
            </View>
          </TouchableOpacity>
        )}
      />
      <View style={styles.bottomNav}>
        <TouchableOpacity onPress={() => setTab('home')}><Text style={[styles.navText, tab === 'home' && styles.navActive]}>Discover</Text></TouchableOpacity>
        <TouchableOpacity onPress={() => setTab('fav')}><Text style={[styles.navText, tab === 'fav' && styles.navActive]}>Saved ({favs.length})</Text></TouchableOpacity>
        <TouchableOpacity onPress={() => alert('Café Finders — Brunei\nA quiet guide to small cafes.')}><Text style={styles.navText}>About</Text></TouchableOpacity>
      </View>
      <Modal visible={!!selected} animationType="slide">
        <SafeAreaView style={{ flex: 1, backgroundColor: '#FAF7F2' }}>
          <ScrollView showsVerticalScrollIndicator={false}>
            {selected && <>
              <View style={{ paddingHorizontal: 24 }}><TouchableOpacity onPress={() => setSelected(null)}><Text style={{ fontSize: 15, color: '#8A7A6D', marginVertical: 12 }}>← Back</Text></TouchableOpacity></View>
              <Image source={{ uri: selected.img }} style={styles.detailImg} />
              <View style={{ padding: 24 }}>
                <Text style={styles.smallLabel}>{selected.district.toUpperCase()} • {selected.tag.toUpperCase()}</Text>
                <Text style={styles.detailName}>{selected.name}</Text>
                <Text style={styles.detailMeta}>★ {selected.rating} • {selected.price} • {selected.open}</Text>
                <View style={styles.divider} />
                <Text style={styles.detailDesc}>{selected.desc}</Text>
                <Text style={styles.detailAddress}>📍 {selected.address}</Text>
                <Text style={styles.detailAddress}>Halal • Quiet • Local</Text>
                <TouchableOpacity style={styles.btnPrimary} onPress={() => Linking.openURL(`https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(selected.name + ' ' + selected.address)}`)}><Text style={styles.btnPrimaryText}>Get Directions →</Text></TouchableOpacity>
                <TouchableOpacity style={styles.btnGhost} onPress={() => Linking.openURL(`tel:${selected.phone}`)}><Text style={styles.btnGhostText}>Call Cafe</Text></TouchableOpacity>
                <TouchableOpacity style={styles.btnGhost} onPress={() => toggleFav(selected.id)}><Text style={styles.btnGhostText}>{favs.includes(selected.id) ? '♥ Saved to Favourites' : '♡ Save to Favourites'}</Text></TouchableOpacity>
              </View>
            </>}
          </ScrollView>
        </SafeAreaView>
      </Modal>
    </SafeAreaView>
  );
}
